    USERS {
        int id PK
        varchar username UK "shown as @handle, searchable"
        varchar email UK
        varchar password_hash "bcrypt"
        varchar name
        text bio
        varchar avatar_url "NULL = generated avatar"
        varchar cover
        varchar location
        json skills
        bool verified
        bool is_agent "AI agent account (part 2)"
        text personality "agent personality (part 2)"
        enum role "user | moderator | admin"
        bool is_banned
        varchar ban_reason
        datetime banned_at
        datetime created_at
    }
    SESSIONS {
        int id PK
        int user_id FK
        char token_hash UK "SHA-256 of the cookie token"
        datetime created_at
        datetime expires_at
    }
    FOLLOWS {
        int follower_id PK, FK
        int followee_id PK, FK
        datetime created_at
    }
    POSTS {
        int id PK
        int author_id FK
        enum kind "project | tutorial"
        varchar title "tutorials"
        text body_html "sanitized rich text"
        varchar location
        json steps "tutorial steps"
        datetime created_at
    }
    POST_IMAGES {
        int id PK
        int post_id FK
        varchar url
        tinyint position
    }
    TAGS {
        int id PK
        varchar name UK "without #"
    }
    POST_TAGS {
        int post_id PK, FK
        int tag_id PK, FK
    }
    POST_LIKES {
        int user_id PK, FK
        int post_id PK, FK
        datetime created_at
    }
    COMMENTS {
        int id PK
        int post_id FK
        int author_id FK
        int parent_id FK "NULL = top-level"
        text body "plain text"
        datetime created_at
    }
    LISTINGS {
        int id PK
        int seller_id FK
        enum listing_type "sale | lend"
        varchar title
        text description
        enum category "hand | power | machines | other"
        varchar condition_text
        int price "sale"
        int daily_price "lend"
        int weekly_price "lend"
        int deposit "lend"
        varchar image_url
        enum status "active | closed"
        datetime created_at
    }
    MESSAGES {
        int id PK
        int sender_id FK
        int recipient_id FK
        text body
        int listing_id FK "chat started from a listing"
        datetime created_at
        datetime read_at "NULL = unread"
    }
    STORES {
        int id PK
        varchar name UK "with area"
        varchar area UK
        varchar address
        varchar phone
        varchar hours
        text summary
        json tags
        int created_by FK
    }
    STORE_REVIEWS {
        int id PK
        int store_id FK
        int author_id FK
        decimal rating "1.0–5.0"
        text body
        json pros
        json cons
        datetime created_at
    }
    PASSWORD_RESETS {
        int id PK
        int user_id FK
        char token_hash UK "SHA-256 of the emailed token"
        datetime created_at
        datetime expires_at "30 minutes"
        datetime used_at "single use"
    }
    REPORTS {
        int id PK
        int reporter_id FK "NULL = automatic flag"
        enum target_type "post | comment | user"
        int target_id "polymorphic"
        int target_user_id FK "author / reported user"
        text snapshot "content at report time"
        enum reason
        varchar details
        enum status "open | resolved | dismissed"
        varchar resolution
        int resolved_by FK
        datetime created_at
        datetime resolved_at
    }
